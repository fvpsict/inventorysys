document.addEventListener("DOMContentLoaded", () => { const faultTableBody = document.querySelector("#fault-table tbody"); const faultForm = document.querySelector("#fault-modal-form"); const addFaultBtn = document.getElementById("add-fault-btn"); const searchInput = document.getElementById("search-fault"); const filterSelect = document.getElementById("filter-equipmenttype"); let editingIndex = null; // Load and render data function loadFaults() { const faults = JSON.parse(localStorage.getItem("faultReports") || "[]"); return faults; } function saveFaults(faults) { localStorage.setItem("faultReports", JSON.stringify(faults)); } function renderTable() { const faults = loadFaults(); const searchQuery = searchInput.value.toLowerCase(); const filterType = filterSelect.value.toLowerCase(); faultTableBody.innerHTML = ""; faults.forEach((fault, index) => { if ( (filterType === "all" || fault.EquipmentType.toLowerCase() === filterType) && Object.values(fault).some(value => String(value).toLowerCase().includes(searchQuery) ) ) { const row = document.createElement("tr"); row.innerHTML = ` ${fault.DateReported} ${fault.EquipmentType} ${fault.Equipment} ${fault.BrandModel} ${fault.SerialNumber} ${fault.Location} ${fault.RoomNumber} ${fault.FaultDescription} `; faultTableBody.appendChild(row); } }); attachRowListeners(); } function attachRowListeners() { document.querySelectorAll(".edit-btn").forEach(btn => { btn.addEventListener("click", (e) => { const index = e.target.getAttribute("data-index"); openEditModal(index); }); }); document.querySelectorAll(".delete-btn").forEach(btn => { btn.addEventListener("click", (e) => { const index = e.target.getAttribute("data-index"); deleteFault(index); }); }); document.querySelectorAll(".status-dropdown").forEach((dropdown, index) => { dropdown.addEventListener("change", (e) => { const faults = loadFaults(); faults[index].Status = e.target.value; saveFaults(faults); renderTable(); }); }); } function openEditModal(index) { const faults = loadFaults(); const fault = faults[index]; for (let key in fault) { const field = faultForm.elements[key]; if (field) { field.value = fault[key]; } } editingIndex = index; new bootstrap.Modal(document.getElementById("faultModal")).show(); } function deleteFault(index) { if (confirm("Are you sure you want to delete this entry?")) { const faults = loadFaults(); faults.splice(index, 1); saveFaults(faults); renderTable(); } } faultForm.addEventListener("submit", (e) => { e.preventDefault(); const formData = new FormData(faultForm); const newFault = {}; formData.forEach((value, key) => { newFault[key] = value.trim(); }); const faults = loadFaults(); if (editingIndex !== null) { faults[editingIndex] = newFault; editingIndex = null; } else { faults.push(newFault); } saveFaults(faults); renderTable(); bootstrap.Modal.getInstance(document.getElementById("faultModal")).hide(); faultForm.reset(); }); addFaultBtn.addEventListener("click", () => { editingIndex = null; faultForm.reset(); new bootstrap.Modal(document.getElementById("faultModal")).show(); }); searchInput.addEventListener("input", renderTable); filterSelect.addEventListener("change", renderTable); renderTable(); });
