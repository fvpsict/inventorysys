document.addEventListener("DOMContentLoaded", () => { const inventoryTable = document.querySelector("#inventory-table tbody"); const addItemBtn = document.getElementById("add-item-btn"); const filterSelect = document.getElementById("filter-equipmenttype"); const inventoryForm = document.getElementById("inventory-form"); const modalElement = document.getElementById("inventoryModal"); const modal = new bootstrap.Modal(modalElement); let editingIndex = null; const localStorageKey = "inventoryData"; const allHeaders = [ "EquipmentType", "Vendor", "BrandModel", "Profile", "Custodian", "AssetNo", "SerialNumber", "Location", "EndDate", "StartDate", "Hostname", "SSOE PO Number", "Cart No", "SanitiseDate", "Fault" ]; function loadData() { return JSON.parse(localStorage.getItem(localStorageKey)) || []; } function saveData(data) { localStorage.setItem(localStorageKey, JSON.stringify(data)); } function calculateDuration(startDate, endDate) { const start = new Date(startDate); const end = new Date(endDate); if (isNaN(start) || isNaN(end)) return ""; const diffYears = end.getFullYear() - start.getFullYear(); const diffMonths = end.getMonth() - start.getMonth(); const years = diffYears + (diffMonths < 0 ? -1 : 0); const months = (diffMonths + 12) % 12; return `${years}y ${months}m`; } function renderTable(data) { inventoryTable.innerHTML = ""; data.forEach((item, index) => { const tr = document.createElement("tr"); allHeaders.forEach(header => { const td = document.createElement("td"); td.textContent = item[header] || ""; tr.appendChild(td); }); // Duration in use column const durationTd = document.createElement("td"); durationTd.textContent = calculateDuration(item["StartDate"], item["EndDate"]); tr.appendChild(durationTd); // Placeholder Lamp Hour and DateUpdated const lampHourTd = document.createElement("td"); lampHourTd.textContent = item["Lamp Hour"] || ""; tr.appendChild(lampHourTd); const dateUpdatedTd = document.createElement("td"); dateUpdatedTd.textContent = new Date().toLocaleDateString(); tr.appendChild(dateUpdatedTd); // Actions const actionsTd = document.createElement("td"); const editBtn = document.createElement("button"); editBtn.textContent = "Edit"; editBtn.className = "btn btn-primary btn-sm me-1"; editBtn.onclick = () => openForm(item, index); const deleteBtn = document.createElement("button"); deleteBtn.textContent = "Delete"; deleteBtn.className = "btn btn-danger btn-sm"; deleteBtn.onclick = () => { const confirmDelete = confirm("Delete this entry?"); if (confirmDelete) { const currentData = loadData(); currentData.splice(index, 1); saveData(currentData); applyFilter(); } }; actionsTd.appendChild(editBtn); actionsTd.appendChild(deleteBtn); tr.appendChild(actionsTd); inventoryTable.appendChild(tr); }); } function openForm(item = {}, index = null) { inventoryForm.innerHTML = ""; editingIndex = index; allHeaders.forEach(header => { const formGroup = document.createElement("div"); formGroup.className = "mb-3"; const label = document.createElement("label"); label.className = "form-label"; label.textContent = header; label.setAttribute("for", header); const input = document.createElement("input"); input.className = "form-control"; input.id = header; input.name = header; input.value = item[header] || ""; formGroup.appendChild(label); formGroup.appendChild(input); inventoryForm.appendChild(formGroup); }); const submitBtn = document.createElement("button"); submitBtn.type = "submit"; submitBtn.className = "btn btn-success"; submitBtn.textContent = index === null ? "Add Item" : "Update Item"; inventoryForm.appendChild(submitBtn); modal.show(); } inventoryForm.addEventListener("submit", e => { e.preventDefault(); const formData = {}; allHeaders.forEach(header => { formData[header] = document.getElementById(header).value; }); const data = loadData(); if (editingIndex === null) { data.push(formData); } else { data[editingIndex] = formData; } saveData(data); modal.hide(); applyFilter(); }); addItemBtn.addEventListener("click", () => { openForm(); }); filterSelect.addEventListener("change", applyFilter); function applyFilter() { const filterValue = filterSelect.value; const data = loadData(); if (filterValue === "All") { renderTable(data); } else { const filtered = data.filter(item => item.EquipmentType === filterValue); renderTable(filtered); } } // Initialize applyFilter(); });
