const tbody = document.querySelector("#inventory-table tbody"); const filterSelect = document.getElementById("filter-equipmenttype"); const searchInput = document.getElementById("search-inventory"); const addItemBtn = document.getElementById("add-item-btn"); let inventory = JSON.parse(localStorage.getItem("inventoryData") || "[]"); function calculateDuration(startDate, endDate) { if (!startDate) return ""; const start = new Date(startDate); const end = endDate ? new Date(endDate) : new Date(); if (start > end) return ""; let years = end.getFullYear() - start.getFullYear(); let months = end.getMonth() - start.getMonth(); if (months < 0) { years--; months += 12; } let result = ""; if (years > 0) result += `${years} yr${years > 1 ? "s" : ""} `; if (months > 0) result += `${months} mo${months > 1 ? "s" : ""}`; return result.trim() || "<1 mo"; } function renderTable() { tbody.innerHTML = ""; const filterVal = filterSelect.value.toLowerCase(); const searchVal = searchInput.value.trim().toLowerCase(); const filtered = inventory.filter((item) => { const matchesFilter = filterVal === "all" || (item.equipmentType || "").toLowerCase() === filterVal; const matchesSearch = Object.values(item) .join(" ") .toLowerCase() .includes(searchVal); return matchesFilter && matchesSearch; }); filtered.forEach((item, index) => { const tr = document.createElement("tr"); tr.innerHTML = ` ${renderCell("equipmentType", item.equipmentType, "select")} ${renderCell("equipment", item.equipment, "dropdown")} ${renderCell("vendor", item.vendor)} ${renderCell("brandModel", item.brandModel)} ${renderCell("profile", item.profile)} ${renderCell("custodian", item.custodian)} ${renderCell("assetNo", item.assetNo)} ${renderCell("serialNumber", item.serialNumber)} ${renderCell("location", item.location)} ${renderCell("endDate", item.endDate, "date")} ${renderCell("startDate", item.startDate, "date")} ${renderCell("hostname", item.hostname)} ${renderCell("ssoePoNumber", item.ssoePoNumber)} ${renderCell("cartNo", item.cartNo)} ${renderCell("sanitiseDate", item.sanitiseDate, "date")} ${calculateDuration(item.startDate, item.endDate)} ${item.dateUpdated || ""} `; tbody.appendChild(tr); }); attachDeleteHandlers(); attachChangeHandlers(); } function renderCell(field, value = "", type = "text") { if (type === "date") { return ``; } else if (type === "dropdown" && field === "equipment") { return ` `; } else if (type === "select" && field === "equipmentType") { return ` `; } else { return ``; } } function attachChangeHandlers() { tbody.querySelectorAll("input, select").forEach((el) => { el.addEventListener("change", (e) => { const td = el.closest("td"); const row = el.closest("tr"); const index = Array.from(tbody.children).indexOf(row); const field = el.dataset.field; const value = el.value; inventory[index][field] = value.trim(); inventory[index].dateUpdated = new Date().toLocaleDateString(); const durationCell = row.querySelector(".duration-cell"); durationCell.textContent = calculateDuration( inventory[index].startDate, inventory[index].endDate ); localStorage.setItem("inventoryData", JSON.stringify(inventory)); renderTable(); }); }); } function attachDeleteHandlers() { document.querySelectorAll(".delete-btn").forEach((btn) => { btn.addEventListener("click", () => { const idx = parseInt(btn.dataset.index); if (confirm("Are you sure you want to delete this entry?")) { inventory.splice(idx, 1); localStorage.setItem("inventoryData", JSON.stringify(inventory)); renderTable(); } }); }); } addItemBtn.addEventListener("click", () => { const today = new Date().toLocaleDateString(); inventory.push({ equipmentType: "", equipment: "", vendor: "", brandModel: "", profile: "", custodian: "", assetNo: "", serialNumber: "", location: "", endDate: "", startDate: "", hostname: "", ssoePoNumber: "", cartNo: "", sanitiseDate: "", dateUpdated: today, }); localStorage.setItem("inventoryData", JSON.stringify(inventory)); renderTable(); }); filterSelect.addEventListener("change", renderTable); searchInput.addEventListener("input", renderTable); renderTable();
