document.addEventListener("DOMContentLoaded", () => { const tableBody = document.querySelector("#inventory-table tbody"); const form = document.getElementById("inventory-form"); const modal = new bootstrap.Modal(document.getElementById("inventoryModal")); const addBtn = document.getElementById("add-item-btn"); const equipmentType = document.getElementById("equipmentType"); const equipmentField = document.getElementById("equipmentField"); let editIndex = null; function renderTable(data = getInventoryData()) { tableBody.innerHTML = ""; data.forEach((item, index) => { const row = document.createElement("tr"); row.innerHTML = ` ${item.EquipmentType} ${item.Equipment || ""} ${item.Vendor} ${item.BrandModel} ${item.Profile} ${item.Custodian} ${item.AssetNo} ${item.SerialNumber} ${item.Location} ${item.EndDate} ${item.StartDate} ${item.Hostname} ${item.SSOE_PONumber} ${item.CartNo} ${item.SanitiseDate} ${item.DateUpdated} `; tableBody.appendChild(row); }); } function getInventoryData() { return JSON.parse(localStorage.getItem("inventoryData") || "[]"); } function saveInventoryData(data) { localStorage.setItem("inventoryData", JSON.stringify(data)); } function clearForm() { form.reset(); editIndex = null; equipmentField.classList.remove("is-invalid"); } function fillForm(data) { form.EquipmentType.value = data.EquipmentType; form.Equipment.value = data.Equipment || ""; form.Vendor.value = data.Vendor; form.BrandModel.value = data.BrandModel; form.Profile.value = data.Profile; form.Custodian.value = data.Custodian; form.AssetNo.value = data.AssetNo; form.SerialNumber.value = data.SerialNumber; form.Location.value = data.Location; form.EndDate.value = data.EndDate; form.StartDate.value = data.StartDate; form.Hostname.value = data.Hostname; form["SSOE_PONumber"].value = data.SSOE_PONumber; form.CartNo.value = data.CartNo; form.SanitiseDate.value = data.SanitiseDate; form.DateUpdated.value = new Date().toLocaleDateString("en-GB", { day: "2-digit", month: "long", year: "numeric", }); } addBtn.addEventListener("click", () => { clearForm(); modal.show(); }); equipmentType.addEventListener("change", () => { if (equipmentType.value === "SSOE") { equipmentField.setAttribute("required", true); } else { equipmentField.removeAttribute("required"); equipmentField.classList.remove("is-invalid"); } }); form.addEventListener("submit", (e) => { e.preventDefault(); if (equipmentType.value === "SSOE" && equipmentField.value === "") { equipmentField.classList.add("is-invalid"); return; } const newItem = { EquipmentType: form.EquipmentType.value, Equipment: form.Equipment.value, Vendor: form.Vendor.value, BrandModel: form.BrandModel.value, Profile: form.Profile.value, Custodian: form.Custodian.value, AssetNo: form.AssetNo.value, SerialNumber: form.SerialNumber.value, Location: form.Location.value, EndDate: form.EndDate.value, StartDate: form.StartDate.value, Hostname: form.Hostname.value, SSOE_PONumber: form["SSOE_PONumber"].value, CartNo: form.CartNo.value, SanitiseDate: form.SanitiseDate.value, DateUpdated: new Date().toLocaleDateString("en-GB", { day: "2-digit", month: "long", year: "numeric", }), }; const data = getInventoryData(); if (editIndex !== null) { data[editIndex] = newItem; } else { data.push(newItem); } saveInventoryData(data); renderTable(); modal.hide(); }); tableBody.addEventListener("click", (e) => { const index = e.target.dataset.index; const data = getInventoryData(); if (e.target.classList.contains("edit-btn")) { editIndex = index; fillForm(data[index]); modal.show(); } if (e.target.classList.contains("delete-btn")) { if (confirm("Are you sure you want to delete this item?")) { data.splice(index, 1); saveInventoryData(data); renderTable(); } } }); renderTable(); });
